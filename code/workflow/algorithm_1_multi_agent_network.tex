\begin{algorithm}[H]
\caption{LLM multi-agent workflow for constructing the Citywalk
mobility-theme-affect network}
\label{alg:three-agent-workflow}
\small
\begin{algorithmic}[1]
\setlength{\itemsep}{1.5pt}
\Require Citywalk notes \(X=\{x_i\}_{i=1}^{n}\), LLM model
\texttt{qwen-plus}, LLM temperature \(\tau=0\), Amap API
\(\gamma(\cdot)\), POI aggregation threshold \(r=100\,\mathrm{m}\), and
maximum distance retained between adjacent POIs
\(d_{\max}=1{,}500\,\mathrm{m}\)
\Ensure Directed, weighted geographical network
\(\mathcal{G}=(\mathcal{V},\mathcal{E},w_e)\)

\State Set \(\mathrm{LLM}\gets\text{\texttt{qwen-plus}}\); instantiate
\(\AgentSpatial^{\mathrm{LLM}}\),
\(\AgentTheme^{\mathrm{LLM}}\), and \(\AgentPAD^{\mathrm{LLM}}\) with
specialized roles and structured output schemas; initialize
\(\mathcal{S}\gets\emptyset\)

\ForAll{Citywalk notes \(x_i\in X\)}
    \State \(P_i\gets\AgentSpatial^{\mathrm{LLM}}(x_i)\)
    \Comment{Agent 1: spatial trajectory coding}
    \Statex \hspace{\algorithmicindent}
    \(P_i=\{(\mathrm{location\_index}_{ij},
    \mathrm{location\_POI}_{ij},
    \mathrm{text\_evidence}_{ij})\}_{j=1}^{m_i}\)
    \Comment{structured, chronological array of location points}

    \State \(t_i\gets\AgentTheme^{\mathrm{LLM}}(x_i,P_i)\)
    \Comment{Agent 2: walking theme categorization}
    \Statex \hspace{\algorithmicindent}
    \(t_i=(\mathrm{Primary\_Theme}_i,
    \mathrm{Theme\_Reasoning}_i)\)
    \Comment{one Primary\_Theme representing the dominant focus}

    \State \(A_i\gets\AgentPAD^{\mathrm{LLM}}(x_i,P_i)\)
    \Comment{Agent 3: PAD-based affective appraisal}
    \Statex \hspace{\algorithmicindent}
    \(A_i=\{(\mathrm{P\_score}_{ij},
    \mathrm{A\_score}_{ij},\mathrm{D\_score}_{ij},
    \mathrm{is\_inherited}_{ij},
    \mathrm{reasoning}_{ij})\}_{j=1}^{m_i}\)
    \Comment{affective appraisal for each featured POI}

    \State \(P_i\gets\operatorname{SemanticFilter}(P_i)\)
    \Comment{screen POI names for geocodability}
    \State \(Z_i\gets\gamma(P_i,\text{``Xiamen''})\)
    \Comment{geocoding and coordinate transformation}

    \State \(S_i\gets
    \operatorname{Join}(P_i,Z_i,t_i,A_i)\)
    \Comment{link through shared note and location identifiers}
    \State Append \(S_i\) to \(\mathcal{S}\)
\EndFor

\State \(\displaystyle
\mathcal{V}=\{v_{ij}=c_r(s_{ij}):s_{ij}\in\mathcal{S}\},\qquad
\bar{\mathbf a}_v=\frac{1}{|v|}
\sum_{s_{ij}\in v}\mathbf a_{ij}\)
\Comment{consolidate POIs and average their PAD vectors}
\State \(\displaystyle
\mathcal{E}=\{(v_{ij},v_{i,j+1}):
v_{ij}\ne v_{i,j+1},\
d(v_{ij},v_{i,j+1})
\le d_{\max}\}\)
\Comment{link consecutively reported POIs after spatial aggregation}
\State \(\displaystyle
w_e(u,v)=\sum_{i=1}^{n}\sum_{j=1}^{|S_i|-1}
\mathbb{I}[(v_{ij},v_{i,j+1})=(u,v)]\)
\Comment{cumulative count of linkages between a specific pair of nodes}
\State \(\mathcal{G}\gets(\mathcal{V},\mathcal{E},w_e)\)
\State \Return \(\mathcal{G}\)
\end{algorithmic}
\end{algorithm}
